```typescript
// 流式数据处理
if (getStorageSetting()?.stream) {
  const reader = res.body?.getReader();
  const decoder = new TextDecoder();
  
  // 增量更新UI
  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    
    const text = decoder.decode(value);
    const parsed = parseSSEResponse(text);
    
    // 更新消息内容
    const updatedMessages = [...store.messages];
    updatedMessages[updatedMessages.length - 1] = {
      ...updatedMessages[updatedMessages.length - 1],
      text: parsed.content,
    };
    store.setMessages(updatedMessages);
  }
}

const handleFileChange = async (e: ChangeEvent<HTMLInputElement>) => {
  const files = e.target.files;
  if (!files?.length) return;
  
  // 文件类型验证
  const file = files[0];
  if (!allowFileList.includes(file.type)) {
    toast.error("不支持的文件类型");
    return;
  }
  
  // 文件大小检查
  if (file.size > 1024 * 1024 * 10) {
    toast.error("文件大小不能超过10MB");
    return;
  }
  
  // 准备上传
  const fileInfo: FileInfoInter = {
    file,
    name: file.name,
    status: "uploading",
  };
  
  setFiles([...files, fileInfo]);
  
  // 执行上传
  try {
    const res = await asyncFileUpload(file);
    const data = await res.json();
    
    // 更新文件状态
    const updatedFiles = [...files];
    const index = updatedFiles.findIndex((f) => f.name === file.name);
    updatedFiles[index] = {
      ...updatedFiles[index],
      file_id: data.file_id,
      status: "uploaded",
    };
    setFiles(updatedFiles);
  } catch (error) {
    // 错误处理...
  }
};

// 生成Code Challenge
export function generateCodeChallenge() {
  const codeVerifier = generateCodeVerifier();
  setStorageSetting({
    ...getStorageSetting(),
    code_verifier: codeVerifier,
  });
  
  // 使用SHA-256哈希计算
  const hash = CryptoJS.SHA256(codeVerifier);
  
  // 转换为Base64编码
  const base64 = CryptoJS.enc.Base64.stringify(hash);
  
  // 转换为Base64URL编码
  return base64.replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
}

// OAuth Token获取
export const asyncOAuthToken = async (code: string, code_verifier: string) => {
  return await fetch(`${getCustomProxyUrl()}/v1/oauth/token`, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      client_id: getStorageSetting()?.client_id,
      code,
      code_verifier,
      redirect_uri,
      grant_type: "authorization_code",

  });
};

// Markdown组件
const Markdown: FC<{ children: string }> = ({ children }) => {
  return (
    <ReactMarkdown
      remarkPlugins={[remarkMath, supersub, remarkBreaks, remarkGfm]}
      rehypePlugins={[
        [rehypeHighlight, { detect: false, ignoreMissing: true }],
      ]}
      components={{
        code: ({ node, inline, className, children, ...props }) => {
          const match = /language-(\w+)/.exec(className || "");
          return !inline && match ? (
            <CustomCode className={className}>{children}</CustomCode>
          ) : (
            <code className={className} {...props}>
              {children}
            </code>
          );
        },
        // 其他组件配置...
      }}
    >
      {children}
    </ReactMarkdown>
  );
};

// 代码块组件
function CustomCode(props: { children: ReactNode; className?: string }) {
  const [copied, setCopied] = useState(false);

  const code = useMemo(
    () => reactNodeToString(props.children),
    [props.children]
  );

  useEffect(() => {
    if (copied) {
      setTimeout(() => setCopied(false), 1000);
    }
  }, [copied]);

  return (
    <div className="flex flex-col">
      <div className="text-xs p-2 flex justify-end">
        <CopyToClipboard text={code} onCopy={() => setCopied(true)}>
          <div className="flex flex-row items-center gap-2 cursor-pointer w-fit ml-1">
            <ClipboardDocumentIcon width={20} />
            <span>{copied ? "copied" : "copy code"}</span>
          </div>
        </CopyToClipboard>
      </div>
      <code className={cn(props.className, "px-4")}>{props.children}</code>
    </div>
  );
}
```

## 六、主题切换实现

```typescript
// 主题模式枚举
export enum ThemeMode {
  Light = "light",
  Dark = "dark",
  Auto = "auto",
}

// 应用主题
export function applyThemeMode(mode: ThemeMode) {
  const root = window.document.documentElement;
  
  // 移除现有主题类
  root.classList.remove("light", "dark");
  
  // 根据模式设置主题
  switch (mode) {
    case ThemeMode.Light:
      root.classList.add("light");
      break;
    case ThemeMode.Dark:
      root.classList.add("dark");
      break;
    case ThemeMode.Auto:
      // 根据系统偏好设置主题
      if (window.matchMedia("(prefers-color-scheme: dark)").matches) {
        root.classList.add("dark");
      } else {
        root.classList.add("light");
      }
      break;
  }
  
  // 保存主题设置
  setStorageSetting({
    ...getStorageSetting(),
    theme_mode: mode,
  });
}
```

## 七、错误处理机制

```typescript
// 错误类定义
export class ChatError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "ChatError";
  }
  
  static fromError(error: unknown): ChatError {
    if (error instanceof ChatError) return error;
    
    if (error instanceof Error) {
      return new ChatError(error.message);
    }
    
    return new ChatError("未知错误");
  }
}

// 错误处理示例
try {
  const res = await asyncChat(messages, abort);
  // 处理响应...
} catch (error) {
  console.error("Error:", error);
  const err = ChatError.fromError(error);
  
  // 更新错误状态
  const updatedMessages = [...store.messages];
  updatedMessages[updatedMessages.length - 1] = {
    ...updatedMessages[updatedMessages.length - 1],
    error: err.message,
  };
  store.setMessages(updatedMessages);
